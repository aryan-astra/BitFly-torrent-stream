cmake_minimum_required(VERSION 3.24)
project(BitFly VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
endif()
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
  set(CMAKE_POLICY_DEFAULT_CMP0167 OLD)
endif()
foreach(policy CMP0144 CMP0183)
  if(POLICY ${policy})
    set(CMAKE_POLICY_DEFAULT_${policy} NEW)
  endif()
endforeach()
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Sql Test)
qt_standard_project_setup(REQUIRES 6.8)

set(BITFLY_LIBTORRENT_SOURCE "" CACHE PATH "Optional libtorrent 2.0.11 source checkout")
set(BITFLY_BOOST_ROOT "" CACHE PATH "Optional Boost 1.86.0 source root")
if(BITFLY_BOOST_ROOT)
  set(BOOST_ROOT "${BITFLY_BOOST_ROOT}")
  set(Boost_NO_BOOST_CMAKE ON)
endif()
if(BITFLY_LIBTORRENT_SOURCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(build_tests OFF CACHE BOOL "" FORCE)
  set(build_examples OFF CACHE BOOL "" FORCE)
  set(build_tools OFF CACHE BOOL "" FORCE)
  add_subdirectory("${BITFLY_LIBTORRENT_SOURCE}" external/libtorrent EXCLUDE_FROM_ALL)
  set(BITFLY_TORRENT_TARGET torrent-rasterbar)
else()
  find_package(LibtorrentRasterbar 2.0.11 CONFIG REQUIRED)
  set(BITFLY_TORRENT_TARGET LibtorrentRasterbar::torrent-rasterbar)
endif()
add_library(bitfly_stream STATIC src/stream/HttpServer.cpp src/stream/HttpServer.h)
target_include_directories(bitfly_stream PUBLIC src)
target_link_libraries(bitfly_stream PUBLIC Qt6::Core Qt6::Network)
add_library(bitfly_torrent STATIC src/torrent/TorrentEngine.cpp src/torrent/TorrentEngine.h)
add_library(bitfly_rules STATIC src/torrent/StreamingRules.cpp)
target_include_directories(bitfly_rules PUBLIC src)
target_link_libraries(bitfly_rules PUBLIC Qt6::Core)
target_include_directories(bitfly_torrent PUBLIC src)
target_link_libraries(bitfly_torrent PUBLIC bitfly_rules Qt6::Core Qt6::Network ${BITFLY_TORRENT_TARGET})
add_library(bitfly_platform STATIC src/app/Platform.cpp src/app/SessionStore.cpp
    src/app/ThumbnailArt.cpp src/app/ThumbnailArt.h
    src/app/PreviewCapture.cpp src/app/PreviewCapture.h
    src/app/SearchService.cpp src/app/SearchService.h
    src/app/UpdateService.cpp src/app/UpdateService.h
    src/app/NetworkAssistant.cpp src/app/NetworkAssistant.h)
target_include_directories(bitfly_platform PUBLIC src src/app)
target_link_libraries(bitfly_platform PUBLIC Qt6::Core Qt6::Gui Qt6::Sql Qt6::Network)
if(WIN32)
  target_link_libraries(bitfly_platform PRIVATE crypt32)
endif()
add_library(bitfly_app STATIC src/app/ApplicationController.cpp src/app/ApplicationController.h)
target_compile_definitions(bitfly_app PRIVATE BITFLY_VERSION="${PROJECT_VERSION}")
target_link_libraries(bitfly_app PUBLIC bitfly_stream bitfly_torrent bitfly_platform Qt6::Gui Qt6::Network)
qt_add_executable(BitFly WIN32 src/main.cpp)
target_include_directories(BitFly PRIVATE packaging)
qt_add_resources(BitFly branding PREFIX "/branding" BASE packaging FILES packaging/BitFly.ico)
target_compile_definitions(BitFly PRIVATE BITFLY_VERSION="${PROJECT_VERSION}")
target_link_libraries(BitFly PRIVATE bitfly_app Qt6::Quick Qt6::QuickControls2)
file(GLOB BITFLY_QML CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "qml/*.qml")
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(BitFly URI BitFly VERSION 1.0 QML_FILES ${BITFLY_QML})
file(GLOB BITFLY_ARTWORK CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/qml/assets/*.png")
qt_add_resources(BitFly artwork PREFIX "/qt/qml/BitFly/qml" BASE qml FILES ${BITFLY_ARTWORK})
if(WIN32)
  target_sources(BitFly PRIVATE packaging/BitFly.rc)
endif()
if(BUILD_TESTING)
  foreach(suite http torrent app integration)
    qt_add_executable(${suite}_tests tests/${suite}_tests.cpp)
    set_target_properties(${suite}_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    target_link_libraries(${suite}_tests PRIVATE Qt6::Test)
    if(suite STREQUAL "http")
      target_link_libraries(${suite}_tests PRIVATE bitfly_stream)
    elseif(suite STREQUAL "torrent")
      target_link_libraries(${suite}_tests PRIVATE bitfly_rules)
    elseif(suite STREQUAL "app")
      target_link_libraries(${suite}_tests PRIVATE bitfly_platform Qt6::Gui)
    else()
      target_link_libraries(${suite}_tests PRIVATE bitfly_stream bitfly_torrent)
    endif()
    if(suite STREQUAL "integration")
      add_test(NAME integration COMMAND integration_tests
        realLibtorrentStreamsRangesAndRestores:v1
        realLibtorrentStreamsRangesAndRestores:v1magnet
        realLibtorrentStreamsRangesAndRestores:v2
        realLibtorrentStreamsRangesAndRestores:hybrid)
    else()
      add_test(NAME ${suite} COMMAND ${suite}_tests)
    endif()
    set_tests_properties(${suite} PROPERTIES TIMEOUT 120 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endforeach()
  set_tests_properties(integration PROPERTIES TIMEOUT 300)
  add_test(NAME recovery COMMAND integration_tests
    realLibtorrentStreamsRangesAndRestores:v1-corrupt-resume
    realLibtorrentStreamsRangesAndRestores:v2-corrupt-resume
    realLibtorrentStreamsRangesAndRestores:v1-quota)
  set_tests_properties(recovery PROPERTIES TIMEOUT 300 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME lifecycle COMMAND integration_tests
    reopensRetainedDataWithTheSameIdAndCanRecreatePurgedId
    restoresActiveSessionsFromTheirOriginalStorageRoot
    storageRootChangesKeepOldRootsManagedForSafePurge)
  set_tests_properties(lifecycle PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(services_tests tests/services_tests.cpp)
  target_link_libraries(services_tests PRIVATE bitfly_platform Qt6::Test)
  add_test(NAME services COMMAND services_tests)
  set_tests_properties(services PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(controller_tests tests/controller_tests.cpp)
  target_link_libraries(controller_tests PRIVATE bitfly_app Qt6::Test)
  add_test(NAME controller COMMAND controller_tests)
  set_tests_properties(controller PROPERTIES TIMEOUT 90 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(preview_tests tests/preview_tests.cpp)
  target_link_libraries(preview_tests PRIVATE bitfly_app Qt6::Test)
  target_compile_definitions(preview_tests PRIVATE BITFLY_PREVIEW_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/preview.avi")
  option(BITFLY_TEST_INSTALLED_VLC "Run the real installed-VLC preview test" OFF)
  if(BITFLY_TEST_INSTALLED_VLC)
    add_test(NAME previews COMMAND preview_tests)
    set_tests_properties(previews PROPERTIES TIMEOUT 180 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  qt_add_executable(qml_smoke tests/qml_smoke.cpp)
  target_link_libraries(qml_smoke PRIVATE bitfly_platform Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Test)
  target_compile_definitions(qml_smoke PRIVATE BITFLY_QML_MAIN="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
  add_test(NAME qml_smoke COMMAND qml_smoke)
  set_tests_properties(qml_smoke PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME qml_search COMMAND qml_smoke --scenario=search --page=Search)
  add_test(NAME qml_large_library COMMAND qml_smoke --scenario=large-library)
  set_tests_properties(qml_search qml_large_library PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS BitFly RUNTIME DESTINATION .)
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION .)
